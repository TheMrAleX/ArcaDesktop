cmake_minimum_required(VERSION 3.22)
project(ArcaDesktop VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 OpenGL Sql Concurrent Network DBus)
find_path(MPV_INCLUDE_DIR mpv/client.h REQUIRED)
find_library(MPV_LIBRARY mpv REQUIRED)
add_library(arca-core STATIC src/Library.cpp src/LibraryMetadata.cpp src/LibraryBackup.cpp src/Library.h src/BackupCodec.cpp src/NameParser.cpp src/Catalog.cpp src/TrackChoice.cpp src/Metadata.cpp)
set_target_properties(arca-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(arca-core PUBLIC Qt6::Core Qt6::Sql Qt6::Concurrent Qt6::Network)
target_include_directories(arca-core PUBLIC src)
qt_add_executable(arca src/main.cpp src/ThumbnailProvider.cpp src/NativeVideo.cpp src/NativeVideo.h src/Mpris.cpp src/Mpris.h resources.qrc)
target_include_directories(arca PRIVATE ${MPV_INCLUDE_DIR})
target_link_libraries(arca PRIVATE arca-core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL Qt6::DBus ${MPV_LIBRARY})
target_compile_definitions(arca PRIVATE ARCA_VERSION="${PROJECT_VERSION}")
include(GNUInstallDirs)
install(TARGETS arca RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.TheMrAleX.Arca.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/arca.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.TheMrAleX.Arca.svg)
install(FILES assets/arca.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME io.github.TheMrAleX.Arca.png)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(arca-tests tests/CoreTest.cpp)
    target_include_directories(arca-tests PRIVATE src)
    target_link_libraries(arca-tests PRIVATE Qt6::Test arca-core)
    add_test(NAME core COMMAND arca-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(arca-catalog-tests tests/CatalogTest.cpp)
    target_include_directories(arca-catalog-tests PRIVATE src)
    target_link_libraries(arca-catalog-tests PRIVATE Qt6::Test arca-core)
    add_test(NAME catalog COMMAND arca-catalog-tests)
    qt_add_executable(arca-metadata-tests tests/MetadataTest.cpp)
    target_link_libraries(arca-metadata-tests PRIVATE Qt6::Test arca-core)
    target_compile_definitions(arca-metadata-tests PRIVATE ARCA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME metadata COMMAND arca-metadata-tests)
    qt_add_executable(arca-backup-tests tests/BackupTest.cpp)
    target_link_libraries(arca-backup-tests PRIVATE Qt6::Test arca-core)
    add_test(NAME backup COMMAND arca-backup-tests)
    qt_add_executable(arca-parser-tests tests/ParserParityTest.cpp)
    target_link_libraries(arca-parser-tests PRIVATE Qt6::Test arca-core)
    target_compile_definitions(arca-parser-tests PRIVATE ARCA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME parser-parity COMMAND arca-parser-tests)
endif()
set(CPACK_PACKAGE_NAME arca-desktop)
set(CPACK_PACKAGE_VENDOR TheMrAleX)
set(CPACK_PACKAGE_CONTACT "https://github.com/TheMrAleX/ArcaDesktop/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Arca: biblioteca multimedia local para Linux")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(ARCA_PACKAGE_PLATFORM "Linux" CACHE STRING "Distribution identifier used in package filenames")
set(CPACK_PACKAGE_FILE_NAME "${CPACK_PACKAGE_NAME}-${PROJECT_VERSION}-${ARCA_PACKAGE_PLATFORM}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CPACK_PRE_BUILD_SCRIPTS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/NormalizePermissions.cmake")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION video)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-dialogs, qml6-module-qt5compat-graphicaleffects, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, qml6-module-qtquick-window, libqt6sql6-sqlite, libmpv2, ffmpeg")
set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtdeclarative, qt6-qtbase, qt6-qt5compat, mpv-libs, ffmpeg-free")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/icons /usr/share/icons/hicolor /usr/share/icons/hicolor/scalable /usr/share/icons/hicolor/scalable/apps /usr/share/icons/hicolor/512x512 /usr/share/icons/hicolor/512x512/apps)
include(CPack)
