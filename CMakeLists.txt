cmake_minimum_required(VERSION 3.22)
project(ArcaDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia Sql Concurrent Network)
qt_add_executable(arca src/main.cpp src/Library.cpp src/Library.h src/NameParser.cpp src/NameParser.h resources.qrc)
target_link_libraries(arca PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Sql Qt6::Concurrent Qt6::Network)
target_compile_definitions(arca PRIVATE ARCA_VERSION="${PROJECT_VERSION}")
include(GNUInstallDirs)
install(TARGETS arca RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.TheMrAleX.Arca.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/arca.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.TheMrAleX.Arca.svg)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(arca-tests tests/CoreTest.cpp src/NameParser.cpp src/Library.cpp src/Library.h)
    target_include_directories(arca-tests PRIVATE src)
    target_link_libraries(arca-tests PRIVATE Qt6::Test Qt6::Sql Qt6::Concurrent Qt6::Network)
    add_test(NAME core COMMAND arca-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_PACKAGE_NAME arca-desktop)
set(CPACK_PACKAGE_VENDOR TheMrAleX)
set(CPACK_PACKAGE_CONTACT "https://github.com/TheMrAleX/ArcaDesktop/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Arca: biblioteca multimedia local para Linux")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION video)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-dialogs, qml6-module-qtmultimedia, qml6-module-qtqml-workerscript, qml6-module-qtquick-templates, qml6-module-qtquick-window, libqt6sql6-sqlite, qt6-svg-plugins, ffmpeg")
set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtdeclarative, qt6-qtmultimedia, qt6-qtbase, qt6-qtsvg, ffmpeg-free")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/icons /usr/share/icons/hicolor /usr/share/icons/hicolor/scalable /usr/share/icons/hicolor/scalable/apps)
include(CPack)
